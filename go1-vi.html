# Python khớp với bảng
from collections import defaultdict

def unique_courses(data):
    table = defaultdict(set)          # course → set of learners
    for learner, courses in data.items():
        for course in courses:
            table[course].add(learner)  # Step 2
    answer = []
    for course, people in table.items():
        if len(people) == 1:            # Step 3
            answer.append(course)
    return answer

def easiest_question(answers, responses):
    n = len(answers)
    counts = [0] * n              # Step 2: một ô cho mỗi question
    for row in responses:         # từng learner

from collections import defaultdict, Counter

def most_common_next(sessions):
    table = defaultdict(Counter)   # course → {next: count}
    for session in sessions:
        for i in range(len(session) - 1):   # dừng trước phần tử cuối
            current = session[i]
            nxt = session[i + 1]
            table[current][nxt] += 1
    answer = {}
    for course, counts in table.items():
        winner = counts.most_common(1)[0][0]
        answer[course] = winner
    return answer

range(len(session) - 1) là mẹo: 4 course thì i = 0,1,2 thôi. Index 3 không có next.

Challenge 4 — Concurrent Learners

Họ cho: stream: [user, start, end].

Họ muốn: số stream overlap cùng lúc lớn nhất.

Issue: nhìn một stream không đủ. Overlap là chuyện timeline. Hai stream overlap nếu cái này start trước khi cái kia end.

["User_001", 0, 1000]
["User_002", 500, 2000]
["User_003", 2500, 3000]
["User_004", 400, 1400]

Tại t=500, U1, U2, U4 cùng bật → 3. U3 sau, một mình. Đáp án là 3.

Cách giải không cần vẽ (sweep)

Mỗi stream thành 2 event: start = +1, end = −1. Sort theo time. Đi trái → phải. Nhớ số “đang xem” cao nhất.

TimeEventCurrentPeak
0U1 start +111
400U4 start +122
500U2 start +133
1000U1 end −123
1400U4 end −113
2000U2 end −103
2500U3 start +113
3000U3 end −103
def max_concurrent(streams):
    events = []
    for user, start, end in streams:
        events.append((start, +1))
        events.append((end, -1))
    events.sort()   # time trước; cùng time thì -1 trước +1
    current = 0
    peak = 0
    for time, delta in events:
        current = current + delta
        if current > peak:
            peak = current
    return peak

Trap: (1) cùng user có thể có 2 stream — đếm stream, không đếm user. Dataset 4 max = 4. (2) Cái này end lúc 1000, cái kia start lúc 1000: thường không overlap — để −1 trước +1 khi time bằng nhau (events.sort() làm sẵn vì −1 < +1).

Follow-up: “Peak lúc nào?” Lưu time khi update peak. “Unique users?” Giữ set các user id đang active, peak = max len(set).

import json

Watched Courses — trung bình, điểm là tie

Với mỗi course, next course phổ biến nhất trên mọi session. Item cuối session không có next.

Hỏi trước khi code

  • Tie? Dataset 2, 3, 4 đều có. Nói: “I’ll take first-seen / most_common unless you want alphabetical.”
  • Course_001 → Course_001 có tính không? (Dataset 4. Yes, trừ khi họ bảo skip.)
  • Course không bao giờ có next: bỏ (khớp example).
import json
from collections import defaultdict, Counter

sessions = json.loads('''
[
  ["Course_001", "Course_002", "Course_003", "Course_004"]
]
''')

def most_common_next(sessions):
    transitions = defaultdict(Counter)
    for session in sessions:
        for current, nxt in zip(session, session[1:]):
            transitions[current][nxt] += 1
    return {
        course: counts.most_common(1)[0][0]
        for course, counts in transitions.items()
    }

print(most_common_next(sessions))
# {'Course_001': 'Course_002', 'Course_002': 'Course_003', 'Course_003': 'Course_004'}

Dataset 2 làm tay (nên nói to):

  • Course_001 → 002 một lần và 003 một lần → tie
  • Course_002 → 003 hai lần → Course_003
  • Course_003 và Course_004 → mỗi next một lần → tie

Họ cần bạn thấy tie, không cần rule “đúng tuyệt đối.”

Concurrent Learners — khó nhất, “CS” nhất

Số stream overlap tối đa. Đếm stream, không đếm unique user. Dataset 4 cùng user hai stream một lúc.

Timeline dataset 2 (max = 3):

0         400  500        1000   1400      2000     2500
U1 |----------------------|
U4      |----------------------|
U2           |---------------------------|
U3                                               |------|
         overlap of 3 ───────────

Hỏi trước khi code

  • Cái này end 1000, cái kia start 1000 — có overlap không? Thường không — xử lý end event trước start event khi cùng time.
  • Cùng user, hai stream overlap → đếm 2?

Có thể bắt đầu brute: “với mỗi stream, đếm bao nhiêu stream khác overlap” (O(n²)). Rồi nói mình có thể sweep.

import json

streams = json.loads('''
[
  ["User_001", 0, 1000],
  ["User_002", 500, 2000],
  ["User_003", 2500, 3000],
  ["User_004", 400, 1400]
]
''')

def max_concurrent(streams):
    events = []
    for _user, start, end in streams:
        events.append((start, 1))   # start = +1
        events.append((end, -1))    # end   = -1
    # sort theo time; cùng time thì -1 trước +1
    events.sort(key=lambda e: (e[0], e[1]))
    current = peak = 0
    for _time, delta in events:
        current += delta
        peak = max(peak, current)
    return f"The maximum number of concurrent streams is {peak}."

print(max_concurrent(streams))
# The maximum number of concurrent streams is 3.

Sort key: event là (time, delta). delta là -1 hoặc +1, nên lúc 1000 thì end (-1) đứng trước start (+1). Không overlap.
